package servlet; import com.google.gson.Gson; import conection.Conexao; import jakarta.servlet.annotation.WebServlet; import jakarta.servlet.http.*; import java.io.*; import java.sql.*; @WebServlet("/login") public class LoginServlet extends HttpServlet { private static final long serialVersionUID = 1L; private Gson gson = new Gson(); // POST - realizar login @Override protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws IOException { resp.setContentType("application/json"); resp.setCharacterEncoding("UTF-8"); String email = req.getParameter("email"); String senha = req.getParameter("senha"); if (email == null || senha == null || email.isEmpty() || senha.isEmpty()) { resp.setStatus(400); resp.getWriter().write("{\"erro\":\"Email e senha são obrigatórios!\"}"); return; } try (Connection conn = Conexao.conectar()) { String sql = "SELECT id, nome, email FROM cliente WHERE email = ? AND senha = ?"; PreparedStatement ps = conn.prepareStatement(sql); ps.setString(1, email); ps.setString(2, senha); ResultSet rs = ps.executeQuery(); if (rs.next()) { // Login bem sucedido String json = "{\"id\":" + rs.getInt("id") + ",\"nome\":\"" + rs.getString("nome") + "\"" + ",\"email\":\"" + rs.getString("email") + "\"}"; resp.getWriter().write(json); } else { resp.setStatus(401); resp.getWriter().write("{\"erro\":\"Email ou senha incorretos!\"}"); } } catch (Exception e) { resp.setStatus(500); resp.getWriter().write("{\"erro\":\"" + e.getMessage() + "\"}"); } } }
